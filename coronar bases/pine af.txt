float op_cf       = request.financial(syminfo.tickerid, "CASH_F_OPERATING_ACTIVITIES","TTM", ignore_invalid_symbol=true)
float da          = request.financial(syminfo.tickerid, "DEPRECIATION_AMORTIZATION_CASH_FLOW", "TTM", ignore_invalid_symbol=true)
float equity_earnings = request.financial(syminfo.tickerid, "EQUITY_IN_EARNINGS",    "TTM", ignore_invalid_symbol=true)
float delta_recv  = request.financial(syminfo.tickerid, "CHANGE_IN_RECEIVABLES",     "TTM", ignore_invalid_symbol=true)

float gross_mar_tv_raw  = request.financial(syminfo.tickerid, "GROSS_MARGIN",  "TTM", ignore_invalid_symbol=true)

float ebitda_mar_tv_raw = request.financial(syminfo.tickerid, "EBITDA_MARGIN", "TTM", ignore_invalid_symbol=true)
// FCF histórico anual
float op_cf_fy_cur = request.financial(syminfo.tickerid, "CASH_F_OPERATING_ACTIVITIES", "FY", ignore_invalid_symbol=true)
float op_cf_fy_1   = request.financial(syminfo.tickerid, "CASH_F_OPERATING_ACTIVITIES", "FY", ignore_invalid_symbol=true)[1]
float capex_fy_cur = request.financial(syminfo.tickerid, "CAPITAL_EXPENDITURES", "FY", ignore_invalid_symbol=true)

// ══════════════════════════════════════════════════════════════
f_norm(v) =>
    na(v) ? na : (math.abs(v) > 100.0 ? na : (math.abs(v) > 10.0 ? v / 100.0 : v))
float gross_mar_tv  = f_norm(gross_mar_tv_raw)
float net_mar_tv    = f_norm(net_mar_tv_raw)
float ebitda_mar_tv = f_norm(ebitda_mar_tv_raw)
// ══════════════════════════════════════════════════════════════
// DETECCIÓN DE MONEDA DEL EPS + MISMATCH + FACTOR FINAL
// ══════════════════════════════════════════════════════════════

// Heurística: si el PER con precio convertido es razonable (2-80), el EPS está en USD
// Si el PER con precio directo (close) es razonable, el EPS está en ARS
float pe_test_usd = not na(eps_raw_tv) and eps_raw_tv > 0 and not na(precio_preliminar) ? precio_preliminar / eps_raw_tv : na
float pe_test_ars = not na(eps_raw_tv) and eps_raw_tv > 0 and not na(close) ? close / eps_raw_tv : na
bool eps_parece_usd  = not na(pe_test_usd) and pe_test_usd > 2.0  and pe_test_usd < 80.0
bool eps_parece_ars  = not na(pe_test_ars) and pe_test_ars > 2.0  and pe_test_ars < 80.0

// Resolución de moneda EPS (solo para BYMA ARS; otros instrumentos = "native")
string eps_moneda = es_accion_byma ? (eps_parece_usd and not eps_parece_ars ? "USD" : eps_parece_ars and not eps_parece_usd ? "ARS" : eps_parece_usd and eps_parece_ars ? "ambig" : "indet") : "native"

// ── MISMATCH REAL ──────────────────────────────────────────
// Solo hay mismatch si: precio ARS + financials USD (o viceversa)
hay_mismatch_moneda := es_accion_byma and eps_moneda == "USD"

// ── FACTOR FINAL DE CONVERSIÓN ─────────────────────────────
// Regla a) BYMA ARS + financials ARS → factor = 1.0 (PER directo)
// Regla b) BYMA ARS + financials USD → factor = 1/TCN
// Regla c) USD-D → factor = 1.0 (precio y financials en USD)
// Regla d) CEDEAR → factor = adr_ratio / TCN
// Regla e) ADR / USA → factor = 1.0
float factor_precio_a_financials = 1.0
if es_accion_byma and eps_moneda == "ARS"
    // BYMA ARS con financials ARS: sin conversión, PER directo
    factor_precio_a_financials := 1.0
else if es_accion_byma and eps_moneda == "USD"
    // BYMA ARS con financials USD: necesita conversión por TCN
    factor_precio_a_financials := not na(tcn_proxy) and tcn_proxy > 0 ? 1.0 / tcn_proxy : 1.0
else if es_accion_byma and eps_moneda == "ambig"
    // Ambiguo: si PER directo es razonable, asumir ARS (sin conversión)
    factor_precio_a_financials := eps_parece_ars ? 1.0 : (not na(tcn_proxy) and tcn_proxy > 0 ? 1.0 / tcn_proxy : 1.0)
else if es_cedear_byma
    // CEDEAR: precio ARS, financials USD → (Precio/TCN)*Ratio / EPS
    factor_precio_a_financials := not na(tcn_proxy) and tcn_proxy > 0 ? adr_ratio_final / tcn_proxy : 1.0
else
    // Otros (ADR, USD-D, USA): precio y financials en misma moneda
    factor_precio_a_financials := 1.0

// Precio comparable final (para cálculo de múltiplos)
float precio_comparable = close * factor_precio_a_financials

// EPS normalizado a USD
float eps_normalizado = eps_moneda == "ARS" and not na(tcn_proxy) and tcn_proxy > 0 ? eps_raw_tv / tcn_proxy : (eps_moneda == "USD" or eps_moneda == "native" or eps_moneda == "ambig") ? eps_raw_tv : na

// Sufijo de advertencia para EPS
string eps_moneda_sufijo = eps_moneda == "ARS" ? "÷TCN" : eps_moneda == "ambig" ? "⚠mon" : eps_moneda == "indet" ? "⚠?" : ""
// ══════════════════════════════════════════════════════════════
// RESOLUCIÓN DINÁMICA DE EBITDA
// ══════════════════════════════════════════════════════════════
// Márgenes calculados manualmente para validación
float mar_bruto_m   = not na(tot_revenue) and tot_revenue > 0 ? gross_profit / tot_revenue : na
float mar_neto_m    = not na(tot_revenue) and tot_revenue > 0 ? net_income   / tot_revenue : na
float mar_ebitda_m  = not na(tot_revenue) and tot_revenue > 0 ? ebitda_tv    / tot_revenue : na
float mar_bruto_mn  = f_norm(mar_bruto_m)
float mar_neto_mn   = f_norm(mar_neto_m)
float mar_ebitda_mn = f_norm(mar_ebitda_m)
// Diferencias TV vs calculado
float diff_eb = na(ebitda_mar_tv) or na(mar_ebitda_mn) ? na : math.abs(ebitda_mar_tv - mar_ebitda_mn)
float diff_gm = na(gross_mar_tv)  or na(mar_bruto_mn)  ? na : math.abs(gross_mar_tv  - mar_bruto_mn)
float diff_nm = na(net_mar_tv)    or na(mar_neto_mn)   ? na : math.abs(net_mar_tv    - mar_neto_mn)
bool eb_tv_ok      = not na(ebitda_mar_tv) and math.abs(ebitda_mar_tv) < 1.0
bool gm_tv_ok      = not na(gross_mar_tv)  and math.abs(gross_mar_tv)  < 1.05
bool nm_tv_ok      = not na(net_mar_tv)    and math.abs(net_mar_tv)    < 0.80
bool eb_m_ok       = not na(mar_ebitda_mn) and math.abs(mar_ebitda_mn) < 1.0
bool gm_m_ok       = not na(mar_bruto_mn)  and math.abs(mar_bruto_mn)  < 1.05
bool eb_consistente = na(diff_eb) ? false : diff_eb < 0.15
bool gm_consistente = na(diff_gm) ? false : diff_gm < 0.15
bool nm_consistente = na(diff_nm) ? false : diff_nm < 0.15
// VPP material
float eq_earn_abs = not na(equity_earnings) ? math.abs(equity_earnings) : 0.0
bool tiene_vpp_material = not na(equity_earnings) and not na(ebit) and math.abs(ebit) > 0 and eq_earn_abs / math.abs(ebit) > 0.15
// Proporción VPP → EBITDA según sector
float vpp_a_ebitda_ratio = is_financial ? 0.0 : (sector_regulado_auto ? 0.65 : 0.50)
// Contribución VPP al EBITDA ajustado
float vpp_contrib = tiene_vpp_material and equity_earnings > 0 ? equity_earnings * vpp_a_ebitda_ratio : 0.0
// Efecto macro sobre margen
float efecto_macro = not na(diferencial_forward) ? math.max(-0.15, math.min(0.15, diferencial_forward * (-0.003))) : 0.0
// ── Resolución nivel a nivel ─────────────────────────────────
var float ebitda_resuelto = na

bool tv_ebitda_plausible = not na(tv_margin_check) and tv_margin_check > 0.0 and tv_margin_check < 0.95
if tv_ebitda_plausible and eb_consistente and not tiene_vpp_material
    ebitda_resuelto := ebitda_tv
    ebitda_nivel    := 1
    ebitda_fuente   := "TV"
// Nivel 2: Reconstruido desde EBIT + DA
else if not na(ebit) and not na(da)
    float ebitda_recalc = ebit + da
    float rec_m = not na(tot_revenue) and tot_revenue > 0 ? ebitda_recalc / tot_revenue : na
    if not na(rec_m) and rec_m > 0.0 and rec_m < 0.95
        ebitda_resuelto := ebitda_recalc
        ebitda_nivel    := 2
        ebitda_fuente   := "≈EBIT+DA"
// Nivel 3a: TV + VPP dinámico + macro (holdings con afiliadas)
// [FIX PAM] Permitir VPP aunque Level 1/2 hayan fallado — empresas con grandes
// participaciones por VPP (PAM, GGAL, etc.) distorsionan EBITDA TV crudo
if (ebitda_nivel >= 1 or tiene_vpp_material) and not na(ebitda_tv)
    float base = not na(ebitda_resuelto) ? ebitda_resuelto : ebitda_tv
    ebitda_resuelto := (base + vpp_contrib) * (1.0 + efecto_macro)
    ebitda_nivel    := 3
    string vpp_str  = not na(equity_earnings) ? "VPP" : ""
    string embi_str = not na(embi_raw) ? "+EMBI" : ""
    string macro_str = not na(diferencial_forward) ? "+macro" : ""
    ebitda_fuente   := "~" + vpp_str + embi_str + macro_str
// Nivel 3b: Solo ajuste macro para empresa ARG sin afiliadas relevantes
else if ebitda_nivel == 1 and is_arg_company and not na(diferencial_forward) and math.abs(diferencial_forward) > 10.0
    ebitda_resuelto := ebitda_tv * (1.0 + efecto_macro)
    ebitda_nivel    := 3
    ebitda_fuente   := "~macro"
// Nivel 4: Estimación por margen sectorial
else if ebitda_nivel == 5 and not na(tot_revenue)
    float margen_est = is_financial ? 0.35 : (sector_regulado_auto ? 0.40 : (is_arg_energy_expansion ? 0.45 : 0.25))
    ebitda_resuelto := tot_revenue * margen_est
    ebitda_nivel    := 4
    ebitda_fuente   := "?sect"
// Validación de rango del factor resultante
float ajuste_ebitda_factor = not na(ebitda_resuelto) and not na(ebitda_tv) and math.abs(ebitda_tv) > 0 ? ebitda_resuelto / ebitda_tv : 1.0
bool factor_razonable = ajuste_ebitda_factor > 0.7 and ajuste_ebitda_factor < 4.0
float ebitda = factor_razonable ? ebitda_resuelto : ebitda_tv
string ebitda_fuente_final = factor_razonable ? ebitda_fuente : "TV!rng"
int    ebitda_nivel_final  = factor_razonable ? ebitda_nivel  : 2
float  ajuste_ebitda_final = factor_razonable ? ajuste_ebitda_factor : 1.0
// [FIX PAM] Si EBITDA final es negativo con margen implícito < -10% para una
// empresa que opera con margen bruto positivo, el dato está roto → forzar N/A
// para no contaminar EVA, DCF, D/EB, ROIC con un número falso en la tabla.
bool ebitda_absurdo = not na(ebitda) and ebitda < 0 and not na(tot_revenue) and tot_revenue > 0 and (ebitda / tot_revenue) < -0.10 and not na(gross_profit) and gross_profit > 0
if ebitda_absurdo

    ebitda_fuente_final := "N/D⚠"
    ebitda_nivel_final  := 5

    last_earn_act := earn_actual
    float sur_q = not na(earn_estimate) and earn_estimate != 0.0 ? (earn_actual - earn_estimate) / math.abs(earn_estimate) * 100.0 : na
    float price_react = not na(close) and not na(close[1]) and close[1] != 0 ? (close - close[1]) / close[1] * 100.0 : na
    array.unshift(earn_act_hist,   earn_actual)
    array.unshift(earn_est_hist,   earn_estimate)
    array.unshift(earn_sur_hist,   sur_q)

    if array.size(earn_act_hist) > 32
        array.pop(earn_act_hist)
        array.pop(earn_est_hist)
        array.pop(earn_sur_hist)
        array.pop(price_react_hist)

    if earn_actual > earn_estimate
        eps_hits_total += 1
    eps_total_reports += 1
    float cur_sur = array.size(earn_sur_hist) > 0 ? array.get(earn_sur_hist, 0) : na
    if not na(cur_sur)
        eps_avg_surprise := ((eps_avg_surprise * (eps_total_reports - 1)) + cur_sur) / eps_total_reports
    float cur_react = array.size(price_react_hist) > 0 ? array.get(price_react_hist, 0) : na
    if not na(cur_react)
        eps_volatility_avg := ((eps_volatility_avg * eps_volatility_samples) + math.abs(cur_react)) / (eps_volatility_samples + 1)
        eps_volatility_samples += 1
// Reutilizar earn_estimate para next_eps_estimate (misma llamada)
if not na(earn_estimate) and (na(next_eps_estimate) or math.abs(earn_estimate - next_eps_estimate) > 0.0001)
    next_eps_estimate := earn_estimate
if not na(earn_actual) and (na(last_earn_act) or math.abs(earn_actual - last_earn_act) > 0.0001)
    last_earnings_bar := bar_index
int bars_since_last = not na(last_earnings_bar) ? bar_index - last_earnings_bar : na
