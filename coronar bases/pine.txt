// ══════════════════════════════════════════════════════════════

f_norm(v) =>
    na(v) ? na : (math.abs(v) > 100.0 ? na : (math.abs(v) > 10.0 ? v / 100.0 : v))

float gross_mar_tv  = f_norm(gross_mar_tv_raw)

bool eps_parece_ars  = not na(pe_test_ars) and pe_test_ars > 2.0  and pe_test_ars < 80.0

// EPS normalizado a USD
float eps_normalizado = eps_moneda == "ARS" and not na(tcn_proxy) and tcn_proxy > 0 ? eps_raw_tv / tcn_proxy : (eps_moneda == "USD" or eps_moneda == "native" or eps_moneda == "ambig") ? eps_raw_tv : na

// Sufijo de advertencia para EPS
string eps_moneda_sufijo = eps_moneda == "ARS" ? "÷TCN" : eps_moneda == "ambig" ? "⚠mon" : eps_moneda == "indet" ? "⚠?" : ""

// ══════════════════════════════════════════════════════════════
// RESOLUCIÓN DINÁMICA DE EBITDA
// ══════════════════════════════════════════════════════════════

// Márgenes calculados manualmente para validación
float mar_bruto_m   = not na(tot_revenue) and tot_revenue > 0 ? gross_profit / tot_revenue : na
float mar_neto_m    = not na(tot_revenue) and tot_revenue > 0 ? net_income   / tot_revenue : na
float mar_ebitda_m  = not na(tot_revenue) and tot_revenue > 0 ? ebitda_tv    / tot_revenue : na
float mar_bruto_mn  = f_norm(mar_bruto_m)
float mar_neto_mn   = f_norm(mar_neto_m)
float mar_ebitda_mn = f_norm(mar_ebitda_m)

// Diferencias TV vs calculado
float diff_eb = na(ebitda_mar_tv) or na(mar_ebitda_mn) ? na : math.abs(ebitda_mar_tv - mar_ebitda_mn)
float diff_gm = na(gross_mar_tv)  or na(mar_bruto_mn)  ? na : math.abs(gross_mar_tv  - mar_bruto_mn)
float diff_nm = na(net_mar_tv)    or na(mar_neto_mn)   ? na : math.abs(net_mar_tv    - mar_neto_mn)

bool eb_tv_ok      = not na(ebitda_mar_tv) and math.abs(ebitda_mar_tv) < 1.0
bool gm_tv_ok      = not na(gross_mar_tv)  and math.abs(gross_mar_tv)  < 1.05
bool nm_tv_ok      = not na(net_mar_tv)    and math.abs(net_mar_tv)    < 0.80
bool eb_m_ok       = not na(mar_ebitda_mn) and math.abs(mar_ebitda_mn) < 1.0
bool gm_m_ok       = not na(mar_bruto_mn)  and math.abs(mar_bruto_mn)  < 1.05
bool eb_consistente = na(diff_eb) ? false : diff_eb < 0.15
bool gm_consistente = na(diff_gm) ? false : diff_gm < 0.15
bool nm_consistente = na(diff_nm) ? false : diff_nm < 0.15

// VPP material
float eq_earn_abs = not na(equity_earnings) ? math.abs(equity_earnings) : 0.0
bool tiene_vpp_material = not na(equity_earnings) and not na(ebit) and math.abs(ebit) > 0 and eq_earn_abs / math.abs(ebit) > 0.15

// Proporción VPP → EBITDA según sector
float vpp_a_ebitda_ratio = is_financial ? 0.0 : (sector_regulado_auto ? 0.65 : 0.50)

// Contribución VPP al EBITDA ajustado
float vpp_contrib = tiene_vpp_material and equity_earnings > 0 ? equity_earnings * vpp_a_ebitda_ratio : 0.0

// Efecto macro sobre margen
float efecto_macro = not na(diferencial_forward) ? math.max(-0.15, math.min(0.15, diferencial_forward * (-0.003))) : 0.0

// ── Resolución nivel a nivel ─────────────────────────────────
var float ebitda_resuelto = na
var int   ebitda_nivel    = 5
var string ebitda_fuente  = "N/D"

// Nivel 1: TV directo consistente
float tv_margin_check = not na(ebitda_tv) and not na(tot_revenue) and tot_revenue > 0 ? ebitda_tv / tot_revenue : na
bool tv_ebitda_plausible = not na(tv_margin_check) and tv_margin_check > 0.0 and tv_margin_check < 0.95
if tv_ebitda_plausible and eb_consistente and not tiene_vpp_material
    ebitda_resuelto := ebitda_tv
    ebitda_nivel    := 1
    ebitda_fuente   := "TV"

// Nivel 2: Reconstruido desde EBIT + DA
else if not na(ebit) and not na(da)
    float ebitda_recalc = ebit + da
    float rec_m = not na(tot_revenue) and tot_revenue > 0 ? ebitda_recalc / tot_revenue : na
    if not na(rec_m) and rec_m > 0.0 and rec_m < 0.95
        ebitda_resuelto := ebitda_recalc

float beta_cov_acc = 0.0, beta_ret_acc = 0.0

int   beta_n_pts   = 0
for i = 0 to 251
    float c_i  = close[i]
    float c_i1 = close[i+1]
    float s_i  = spy_close[i]
    float s_i1 = spy_close[i+1]
    if not na(c_i) and not na(c_i1) and c_i1 != 0 and not na(s_i) and not na(s_i1) and s_i1 != 0
        float r_s = (c_i - c_i1) / c_i1
        float r_m = (s_i - s_i1) / s_i1
        beta_cov_acc  += r_s * r_m
        beta_ret_acc  += r_s
        beta_mkt_acc  += r_m
        beta_mkt2_acc += r_m * r_m
        beta_n_pts    += 1

var float pearson_r = na
int n_corr_points = array.size(earn_sur_hist)
if n_corr_points >= 4
    float cx  = 0.0
    float cy  = 0.0
    float cxy = 0.0
    float cx2 = 0.0
    float cy2 = 0.0
    for i = 0 to n_corr_points - 1
        float x = array.get(earn_sur_hist,   i)
        float y = array.get(price_react_hist, i)
        if not na(x) and not na(y)
            cx  += x
            cy  += y
            cxy += x * y
            cx2 += x * x
            cy2 += y * y
    float n   = n_corr_points
    float num = (n * cxy) - (cx * cy)
    float den = math.sqrt(((n * cx2) - (cx * cx)) * ((n * cy2) - (cy * cy)))
    float pr  = den != 0 ? num / den : na
    pearson_r := not na(pr) ? math.max(-1.0, math.min(1.0, pr)) : na

float r_squared   = not na(pearson_r) ? math.max(0.0, math.min(1.0, pearson_r * pearson_r)) : na

bool  corr_positiva = not na(pearson_r) and pearson_r > 0
