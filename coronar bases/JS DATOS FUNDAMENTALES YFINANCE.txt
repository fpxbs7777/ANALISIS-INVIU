    // ========================================
    // 3. OBTENER TOKEN IOL (sin cambios)
    // ========================================
    async obtenerTokenIOL() {
        if (this.tokenCache && !this.isTokenExpired(this.tokenCache)) {
            return this.tokenCache.token;
        }

        try {
            const response = await fetch(`${this.apis.iol.base}/token`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
                body: new URLSearchParams({
                    username: this.apis.iol.credentials.username,
                    password: this.apis.iol.credentials.password,
                    grant_type: 'password'
                })
            });

            if (response.ok) {
                const data = await response.json();
                this.tokenCache = {
                    token: data.access_token,
                    expires: Date.now() + (data.expires_in * 1000)
                };
                console.log('✅ Token IOL obtenido exitosamente');
                return this.tokenCache.token;
            }
        } catch (error) {
            console.error('❌ Error obteniendo token IOL:', error);
        }
        return null;
    }

    isTokenExpired(tokenCache) {
        return !tokenCache || Date.now() > tokenCache.expires;
    }

    // ========================================
    // 4. OBTENER COTIZACIONES IOL (sin cambios)
    // ========================================
    async obtenerCotizaciones(instrumento, pais) {
        const token = await this.obtenerTokenIOL();
        if (!token) return [];

        try {
            const url = `${this.apis.iol.base}/Cotizaciones/${instrumento}/${pais}/Todos`;
            const headers = {
                'Accept': 'application/json',
                'Authorization': `Bearer ${token}`
            };

            const response = await fetch(url, { headers });
            
            if (response.status === 200) {
                const data = await response.json();
                if (data.titulos && data.titulos.length > 0) {
                    console.log(`✅ ${instrumento}/${pais}: ${data.titulos.length} títulos`);
                    return data.titulos;
                } else {
                    console.log(`⚠️ ${instrumento}/${pais}: Sin datos de títulos`);
                    return [];
                }
            } else {
                console.log(`❌ Error en ${instrumento}/${pais}: ${response.status}`);
                return [];
            }
        } catch (error) {
            console.error(`❌ Error procesando ${instrumento}/${pais}:`, error);
            return [];
        }
    }

    async procesarActivoCompleto(titulo, tipoInstrumento, pais, categoria) {
        try {
            // Clasificar el activo
            const clasificacion = this.clasificarActivo(titulo, tipoInstrumento, pais);
            
            // Obtener TODOS los datos fundamentales completos
            const fundamentales = await this.obtenerTodosLosDatosFundamentales(
                clasificacion.tickerYFinance, 
                titulo.simbolo
            );
            
            // Combinar datos
            const activoCompleto = {
                // Datos de IOL
                simboloIOL: titulo.simbolo,
                descripcionIOL: titulo.descripcion || titulo.simbolo,
                precioIOL: titulo.precio,
                variacionIOL: titulo.variacionPorcentual,
                volumenIOL: titulo.volumen,
                monedaIOL: titulo.moneda,
                
                // Clasificación
                tipo: clasificacion.tipo,
                categoria: categoria,
                tickerYFinance: clasificacion.tickerYFinance,
                descripcionCompleta: clasificacion.descripcion,
                
                // Datos fundamentales COMPLETOS
                ...(fundamentales || {}),
                
                // Metadata de procesamiento
                datosFundamentalesDisponibles: !!fundamentales,
                timestampProcesamiento: new Date().toISOString(),
                modulosObtenidos: fundamentales?.totalModulosObtenidos || 0
            };
            
            return activoCompleto;
            
        } catch (error) {
            console.error(`❌ Error procesando ${titulo.simbolo}:`, error);
            return null;
        }
    }

    // ========================================
    // 6. CLASIFICACIÓN DE ACTIVOS (sin cambios)
    // ========================================
    clasificarActivo(titulo, tipoInstrumento, pais) {
        const simbolo = titulo.simbolo || '';
        const descripcion = titulo.descripcion || '';
        
        switch (tipoInstrumento) {
            case 'cedears':
                return {
                    tipo: 'CEDEAR',
                    categoria: 'cedears',
                    tickerYFinance: this.mapearTickerCEDEAR(simbolo),
                    descripcion: `CEDEAR - ${descripcion}`
                };
            
            case 'acciones':
                if (pais === 'argentina') {
                    return {
                        tipo: 'ACCION LOCAL',
                        categoria: 'acciones_locales',
                        tickerYFinance: this.mapearAccionLocal(simbolo),
                        descripcion: `Acción Argentina - ${descripcion}`
                    };
                } else if (pais === 'estados_unidos') {
                    return {
                        tipo: 'ACCION EEUU',
                        categoria: 'acciones_eeuu',
                        tickerYFinance: simbolo,
                        descripcion: `Acción EE.UU - ${descripcion}`
                    };
                }
                break;
            
            case 'adrs':
                return {
                    tipo: 'ADR',
                    categoria: 'adrs',
                    tickerYFinance: simbolo,
                    descripcion: `ADR - ${descripcion}`
                };
        }
        
        return {
            tipo: 'DESCONOCIDO',
            categoria: 'otros',
            tickerYFinance: simbolo,
            descripcion: descripcion
        };
    }

    mapearTickerCEDEAR(simbolo) {
        return simbolo.replace('.BA', '').replace('.D', '').replace('.C', '');
    }

    mapearAccionLocal(simbolo) {
        return simbolo.replace('.BA', '');
    }

        } catch (error) {
            console.error('❌ Error en análisis completo:', error);
            return null;
        }
    }

    async procesarTipoConDatosCompletos(titulos, tipoInstrumento, pais, categoria) {
        console.log(`🔄 Procesando ${categoria} con DATOS COMPLETOS: ${titulos.length} activos`);
        
        const resultados = [];
        
        for (const titulo of titulos) {
            const resultado = await this.procesarActivoCompleto(titulo, tipoInstrumento, pais, categoria);
            if (resultado) {
                resultados.push(resultado);
            }
        }
        
        console.log(`✅ ${categoria} completado: ${resultados.length} activos con datos`);
        return resultados;
    }

    // 8. MOSTRAR RESUMEN COMPLETO
    // ========================================
    mostrarResumenCompleto() {
        console.log('\n📊 ===== ANÁLISIS COMPLETO CON TODOS LOS DATOS FUNDAMENTALES =====');
        
        const tabs = [
            { nombre: 'CEDEARs', data: this.dataframes.cedears },
            { nombre: 'Acciones Locales', data: this.dataframes.acciones_locales },
            { nombre: 'Acciones EE.UU', data: this.dataframes.acciones_eeuu },
            { nombre: 'ADRs', data: this.dataframes.adrs }
        ];
        
        tabs.forEach(tab => {
            console.log(`\n📋 TAB: ${tab.nombre}`);
            console.log(`📊 Total activos: ${tab.data.length}`);
            
            const conDatosCompletos = tab.data.filter(a => a.datosFundamentalesDisponibles);
            console.log(`✅ Con datos completos: ${conDatosCompletos.length}`);
            
            if (conDatosCompletos.length > 0) {
                // Top 3 con más módulos obtenidos
                const topModulos = conDatosCompletos
                    .sort((a, b) => (b.modulosObtenidos || 0) - (a.modulosObtenidos || 0))
                    .slice(0, 3);
                
                console.log('🏆 Top 3 con más datos fundamentales:');
                topModulos.forEach((activo, index) => {
                    console.log(`${index + 1}. ${activo.simboloIOL} - ${activo.nombre || 'Sin nombre'}`);
                    console.log(`   Módulos obtenidos: ${activo.modulosObtenidos || 0}`);
                    console.log(`   Sector: ${activo.sector || 'N/A'}`);
                    console.log(`   ROE: ${activo.returnOnEquity ? (activo.returnOnEquity * 100).toFixed(1) + '%' : 'N/A'}`);
                    console.log(`   PER: ${activo.trailingPE ? activo.trailingPE.toFixed(1) : 'N/A'}`);
                    console.log(`   Beta: ${activo.beta ? activo.beta.toFixed(2) : 'N/A'}`);
                    console.log(`   Analistas: ${activo.numberOfAnalysts || 0}`);
                });
            }
        });
        
        // Estadísticas globales
        const totalActivos = Object.values(this.dataframes).flat().length;
        const totalConDatos = Object.values(this.dataframes)
            .flat()
            .filter(a => a.datosFundamentalesDisponibles).length;
        
        console.log('\n📈 ESTADÍSTICAS GLOBALES:');
        console.log(`📊 Total de activos procesados: ${totalActivos}`);
        console.log(`✅ Con datos fundamentales completos: ${totalConDatos}`);
        console.log(`📊 Tasa de éxito: ${((totalConDatos/totalActivos)*100).toFixed(1)}%`);
        
        console.log('\n🎯 DATOS FUNDAMENTALES DISPONIBLES POR ACTIVO:');
        const datosPorActivo = Object.values(this.dataframes)
            .flat()
            .filter(a => a.datosFundamentalesDisponibles);
        
        if (datosPorActivo.length > 0) {
            const promedioModulos = datosPorActivo.reduce((sum, a) => sum + (a.modulosObtenidos || 0), 0) / datosPorActivo.length;
            console.log(`📊 Promedio de módulos por activo: ${promedioModulos.toFixed(1)}`);
            
            // Verificar disponibilidad de datos clave
            const conROE = datosPorActivo.filter(a => a.returnOnEquity && a.returnOnEquity > 0).length;
            const conPER = datosPorActivo.filter(a => a.trailingPE && a.trailingPE > 0).length;
            const conBeta = datosPorActivo.filter(a => a.beta && a.beta > 0).length;
            const conAnalistas = datosPorActivo.filter(a => a.numberOfAnalysts && a.numberOfAnalysts > 0).length;
            const conDividendos = datosPorActivo.filter(a => a.dividendYield && a.dividendYield > 0).length;
            
            console.log(`📊 Con ROE: ${conROE} (${((conROE/datosPorActivo.length)*100).toFixed(1)}%)`);
            console.log(`📊 Con PER: ${conPER} (${((conPER/datosPorActivo.length)*100).toFixed(1)}%)`);
            console.log(`📊 Con Beta: ${conBeta} (${((conBeta/datosPorActivo.length)*100).toFixed(1)}%)`);
            console.log(`📊 Con analistas: ${conAnalistas} (${((conAnalistas/datosPorActivo.length)*100).toFixed(1)}%)`);
            console.log(`📊 Con dividendos: ${conDividendos} (${((conDividendos/datosPorActivo.length)*100).toFixed(1)}%)`);
        }
        
        console.log('\n✅ Análisis COMPLETO finalizado exitosamente!');
    }

    // ========================================
    // 9. EXPORTACIÓN COMPLETA
    // ========================================
    exportarTabCSV(tabNombre, data, filename) {
        if (!data || data.length === 0) {
            console.log(`⚠️ No hay datos para exportar en ${tabNombre}`);
            return;
        }

        const headers = Object.keys(data[0]);
        const csvContent = [
            headers.join(','),
            ...data.map(activo => 
                headers.map(header => `"${activo[header] || ''}"`).join(',')
            )
        ].join('\n');

        const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
        const link = document.createElement('a');
        link.href = URL.createObjectURL(blob);
        link.download = filename;
        link.click();
        
        console.log(`📁 ${tabNombre} con DATOS COMPLETOS exportado a ${filename}`);
    }

    exportarTodasLasTabsCompletas() {
        const tabs = [
            { nombre: 'CEDEARs', data: this.dataframes.cedears, prefix: 'cedears_completos' },
            { nombre: 'Acciones Locales', data: this.dataframes.acciones_locales, prefix: 'acciones_locales_completas' },
            { nombre: 'Acciones EE.UU', data: this.dataframes.acciones_eeuu, prefix: 'acciones_eeuu_completas' },
            { nombre: 'ADRs', data: this.dataframes.adrs, prefix: 'adrs_completos' }
        ];

        tabs.forEach(tab => {
            this.exportarTabCSV(tab.nombre, tab.data, `${tab.prefix}.csv`);
        });

        // Exportar dataframe combinado completo
        const todosLosActivos = Object.values(this.dataframes).flat();
        this.exportarTabCSV('Todos los Activos - Datos Completos', todosLosActivos, 'todos_los_activos_completos.csv');
    }

        return this.dataframes[tab] || [];
    }

        return this.dataframes;
    }
