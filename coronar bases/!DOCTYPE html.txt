const TODAY = new Date('2026-04-20');
let activeBond = 'gd41';

function fmt(n, d=2) { return n.toLocaleString('es-AR',{minimumFractionDigits:d,maximumFractionDigits:d}); }
function days(dateStr) { return Math.round((new Date(dateStr) - TODAY) / 86400000); }
function fmtFecha(s) {
  const [y,m,d] = s.split('-');
  const meses = ['','Ene','Feb','Mar','Abr','May','Jun','Jul','Ago','Sep','Oct','Nov','Dic'];
  return `${parseInt(d)} ${meses[parseInt(m)]} ${y}`;
}

function selectBond(id, el) {
  activeBond = id;
  document.querySelectorAll('.bc').forEach(c => c.classList.remove('active'));
  el.classList.add('active');
  onMontoChange();
}

function setMonto(v, btn) {
  document.getElementById('monto').value = v;
  document.querySelectorAll('.quick-btn').forEach(b => b.classList.remove('sel'));
  if (btn) btn.classList.add('sel');
  onMontoChange();
}

function onMontoChange() {
  const b = BONDS[activeBond];
  const m = parseFloat(document.getElementById('monto').value) || 0;
  const noms = Math.floor(m / b.precio);
  const chip = document.getElementById('nomsChip');
  if (noms > 0) {
    chip.classList.add('show');
    document.getElementById('nomsVal').textContent = noms.toLocaleString('es-AR') + ' nominales';
    document.getElementById('capitalVal').textContent = 'u$s ' + fmt(noms * b.precio, 0);
  } else { chip.classList.remove('show'); }
}

function showTab(id, btn) {
  document.querySelectorAll('.tab-content').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
  document.getElementById('tab-' + id).classList.add('active');
  btn.classList.add('active');
}

function calcular() {
  const b = BONDS[activeBond];
  const monto = parseFloat(document.getElementById('monto').value) || 0;
  const noms = Math.floor(monto / b.precio);
  if (noms < 1) { alert('El monto ingresado no alcanza para adquirir al menos 1 nominal de este bono.'); return; }

  const invertido = noms * b.precio;
  const VN = b.valorNominal;

  let totalR = 0, totalA = 0, acum = 0;
  const rows = b.flows.map((f, i) => {
    const rUSD = (f.r / 100) * noms * VN * b.valorPar;
    const aUSD = (f.a / 100) * noms * VN * b.valorPar;
    const tot  = rUSD + aUSD;
    acum += tot;
    totalR += rUSD;
    totalA += aUSD;
    return { fecha: f.f, rUSD, aUSD, tot, acum, d: days(f.f) };
  });

  const totalFlow = totalR + totalA;
  const ganancia  = totalFlow - invertido;
  const retPct    = (ganancia / invertido) * 100;
  const proxTot   = rows[0].tot;
  const diasProx  = rows[0].d;

  // Narrative para el cliente
  const narrativa = buildNarrativa(b, noms, invertido, totalFlow, totalR, totalA, ganancia, retPct, rows);

  // Próximos cobros (primeros 4)
  const cobrosHtml = rows.slice(0, 4).map((r, i) => `
    <div class="cobro-chip ${i===0?'next':''}">
      <div class="cobro-date">${fmtFecha(r.fecha)}</div>
      <div class="cobro-amt">u$s ${fmt(r.tot, 0)}</div>
      <div class="cobro-label">${r.d} días · ${i===0?'Próximo cobro':i===1?'2° cobro':i===2?'3° cobro':'4° cobro'}</div>
    </div>
  `).join('');

  // Contexto cards
  const ctxHtml = b.contexto.map(c => `
    <div class="ctx-card">
      <div class="ctx-card-title">
        <div class="ctx-dot ${c.tipo}"></div>
        ${c.titulo}
      </div>
      <div class="ctx-body">${c.texto}</div>
    </div>
  `).join('');

  // Tabla CF
  const tblHtml = rows.map((r, i) => {
    const diasStr = r.d < 0 ? 'Vencido' : r.d === 0 ? 'Hoy' : r.d + ' días';
    return `<tr class="${i===0?'prox':''}">
      <td class="c-ink">${fmtFecha(r.fecha)}${i===0?'<span class="tag-prox">PRÓXIMO</span>':''}</td>
      <td>${diasStr}</td>
      <td class="c-gold">${r.rUSD > 0 ? 'u$s '+fmt(r.rUSD) : '—'}</td>
      <td>${r.aUSD > 0 ? 'u$s '+fmt(r.aUSD) : '—'}</td>
      <td class="c-ink">u$s ${fmt(r.tot)}</td>
      <td>u$s ${fmt(r.acum,0)}</td>
    </tr>`;
  }).join('');

  // Barras
