Función para calcular la correlación y beta del activo respecto de cada factor
def dataframe_factors(seguridad, factores):
code
Code
tickers = [seguridad] + list(factores.keys())

data = download_data(tickers)



# Calcular rendimientos diarios

returns = data.pct_change().dropna()



# Rendimientos del activo y de los factores

seguridad_returns = returns[seguridad]

factores_returns = returns[list(factores.keys())]



rows = []

for factor in factores:

    # Calcular correlación: activo vs factor

    correlacion = np.corrcoef(seguridad_returns, factores_returns[factor])[0, 1]

    # Calcular beta: covarianza / varianza del factor

    covarianza = np.cov(seguridad_returns, factores_returns[factor])[0, 1]

    varianza = np.var(factores_returns[factor])

    beta = covarianza / varianza

    rows.append({'factor': factor, 'correlacion': correlacion, 'beta': beta})



df = pd.DataFrame(rows)

df['sector'] = df['factor'].map(factores)

return df, data

Función para simular un portafolio basado en rendimientos ponderados
def simular_portafolio(data, tickers, cantidades):
code
Code
# Calcular rendimientos diarios de los activos

rendimientos = data[tickers].pct_change().dropna()

# Calcular rendimientos ponderados por las cantidades

pesos = np.array(cantidades) / np.sum(cantidades)

portafolio_rendimientos = rendimientos.dot(pesos)

# Reconstruir el valor del portafolio a partir de los rendimientos acumulados

portafolio = (1 + portafolio_rendimientos).cumprod()

portafolio *= data[tickers].iloc[0].dot(pesos)  # Escalar al valor inicial

return portafolio
