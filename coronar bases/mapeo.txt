import requests
import pandas as pd
import yfinance as yf
import time
import re
import json

# --- CONFIGURACIÓN IOL ---
def obtener_tokens(username, password):
    url = 'https://api.invertironline.com/token'
    payload = {'username': username, 'password': password, 'grant_type': 'password'}
    response = requests.post(url, data=payload)
    if response.status_code == 200:
        return response.json()['access_token']
    return None

def obtener_panel(instrumento, pais, token):
    url = f"https://api.invertironline.com/api/v2/Cotizaciones/{instrumento}/{pais}/Todos"
    headers = {'Authorization': f'Bearer {token}'}
    response = requests.get(url, headers=headers)
    if response.status_code == 200:
        return pd.DataFrame(response.json().get('titulos', []))
    return pd.DataFrame()

# --- LÓGICA DE LIMPIEZA Y MAPEO ---
def extraer_raiz(simbolo):
    return re.sub(r'([A-Z]{2,})[DC]$', r'\1', simbolo)

def mapear_yfinance(simbolo, cedears_limpios):
    correcciones = {
        'VALE.SA': 'VALE', 'TECO.BA': 'TEO', 'TGNO4.BA': 'TGNO4.SA', 'ITUB': 'ITUB4.SA'
    }
    if simbolo in correcciones: return correcciones[simbolo]
    
    raiz = extraer_raiz(simbolo)
    if raiz.endswith(('3', '4', '5', '6', '7', '8', '9')): 
        return raiz + '.SA'
    elif raiz in cedears_limpios: 
        return raiz
    else: 
        return raiz + '.BA'

# --- LÓGICA FINANCIERA BASADA EN TÉCNICAS DE CONTABILIDAD Y FINANZAS PARA DUMMIES Y TÁCTICAS DE MERCADO ---
# Basado en análisis fundamental básico: ratios de liquidez, rentabilidad, endeudamiento
# Sin cálculos de semáforo, solo recopilación de datos crudos para análisis manual

def obtener_datos_fundamentales(info):
    """
    Extrae datos fundamentales clave según técnicas de libros mencionados.
    - Ratios de liquidez (currentRatio)
    - Rentabilidad (ROE, ROA)
    - Endeudamiento (debtToEquity)
    - Valoración (P/E, P/B)
    - Dividendos (dividendYield)
    """
    return {
        'liquidez': {
            'currentRatio': info.get('currentRatio'),
            'quickRatio': info.get('quickRatio')
        },
        'rentabilidad': {
            'returnOnEquity': info.get('returnOnEquity'),
            'returnOnAssets': info.get('returnOnAssets'),
            'profitMargins': info.get('profitMargins')
        },
        'endeudamiento': {
            'debtToEquity': info.get('debtToEquity'),
            'totalDebt': info.get('totalDebt'),
            'totalCash': info.get('totalCash')
        },
        'valoracion': {
            'forwardPE': info.get('forwardPE'),
            'priceToBook': info.get('priceToBook'),
            'enterpriseValue': info.get('enterpriseValue')
        },
        'dividendos': {
            'dividendYield': info.get('dividendYield'),
            'dividendRate': info.get('dividendRate')
        }
    }
