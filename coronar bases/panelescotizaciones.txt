import requests
import pandas as pd
from tokens import obtener_tokens, refrescar_token

# Credenciales de usuario (usar variables de entorno)
import os
username = os.environ.get("IOL_USERNAME", "")
password = os.environ.get("IOL_PASSWORD", "")

# Obtener los tokens una sola vez
bearer_token, refresh_token = obtener_tokens(username, password)
if not bearer_token or not refresh_token:
    raise Exception('Error al obtener los tokens')

def obtener_cotizaciones(instrumento, pais):
    url = f"https://api.invertironline.com/api/v2/Cotizaciones/{instrumento}/{pais}/Todos"
    headers = {
        'Accept': 'application/json',
        'Authorization': f'Bearer {bearer_token}'
    }
    response = requests.get(url, headers=headers)
    if response.status_code == 200:
        data = response.json()
        if 'titulos' in data:
            return pd.DataFrame(data['titulos'])
        else:
            print('No se encontraron datos de títulos en la respuesta.')
            return pd.DataFrame()
    else:
        print(f'Error en la solicitud: {response.status_code}')
        print(response.text)
        return pd.DataFrame()
